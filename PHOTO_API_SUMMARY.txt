QUICK START FOR INTEGRATION
================================================================================

For each photo manager:

1. Replace imports:
   import { useAdminPhotos } from "../../hooks/useAdminPhotos";
   import { useAdminPhotoConfig } from "../../hooks/useAdminPhotoConfig";
   import { UploadPhotoDialog } from "./UploadPhotoDialog";

2. Add state:
   const [isUploadOpen, setIsUploadOpen] = useState(false);

3. Add hooks:
   const { uploadAndAssign, deletePhoto, reorderPhotos, loading, error, clearError } = useAdminPhotos();
   const photoConfig = useAdminPhotoConfig();

4. Create handlers:
   const handleUploadPhoto = async (file: File, altText: string) => {
     await uploadAndAssign(file, CONFIG_KEY, displayOrder, altText);
   };

5. Update PhotoGridManager:
   <PhotoGridManager
     onAddPhoto={() => setIsUploadOpen(true)}
     onRemovePhoto={handleRemovePhoto}
     onReorderPhotos={handleReorderPhotos}
   />

6. Add UploadPhotoDialog:
   <UploadPhotoDialog
     isOpen={isUploadOpen}
     isLoading={loading}
     onUpload={handleUploadPhoto}
     onClose={() => setIsUploadOpen(false)}
     errorMessage={error}
     onClearError={clearError}
   />
